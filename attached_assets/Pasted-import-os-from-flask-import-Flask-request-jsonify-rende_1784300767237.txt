def get_macro_data():
    try:
        fred = Fred(api_key=FRED_API_KEY)
        gdp = fred.get_series('GDP').iloc[-1]
        real_gdp = fred.get_series('GDPC1').iloc[-1]
        cpi = fred.get_series('CPIAUCSL').iloc[-1]
        unemployment = fred.get_series('UNRATE').iloc[-1]
        fed_funds = fred.get_series('FEDFUNDS').iloc[-1]
        cpi_series = fred.get_series('CPIAUCSL')
        inflation_yoy = ((cpi_series.iloc[-1] / cpi_series.iloc[-13]) - 1) * 100
        return {
            "GDP": f"${gdp/1000:.2f} Trillion",
            "Real GDP": f"${real_gdp/1000:.2f} Trillion",
            "CPI": f"{cpi:.2f}",
            "Inflation (YoY)": f"{inflation_yoy:.2f}%",
            "Unemployment Rate": f"{unemployment:.2f}%",
            "Fed Funds Rate": f"{fed_funds:.2f}%"
        }
    except:
        return {"Error": "FRED unavailable"}

def get_breadth_liquidity():
    try:
        tnx = yf.Ticker("^TNX").history(period="5d")['Close'].iloc[-1]
        fvx = yf.Ticker("^FVX").history(period="5d")['Close'].iloc[-1]
        yield_spread = tnx - fvx
        vix = yf.Ticker("^VIX").history(period="5d")['Close'].iloc[-1]
        mega = ["AAPL","MSFT","GOOGL","AMZN","NVDA","META","TSLA","BRK-B","JPM","JNJ"]
        above = 0
        for t in mega:
            try:
                d = yf.Ticker(t).history(period="3mo")
                if len(d)>=50 and d['Close'].iloc[-1] > d['Close'].rolling(50).mean().iloc[-1]:
                    above += 1
            except: pass
        pct = round((above/len(mega))*100,1)
        if pct>70 and vix<20:
            bstat = "🟢 Healthy"
            lstat = "🟢 Normal"
        elif pct<30 or vix>30:
            bstat = "🔴 Weak"
            lstat = "🔴 Tight"
        else:
            bstat = "🟡 Neutral"
            lstat = "🟡 Neutral"
        return {
            "Stocks Above 50-day MA": f"{pct}%",
            "Breadth Status": bstat,
            "VIX": f"{vix:.2f}",
            "10Y Yield": f"{tnx:.2f}%",
            "Yield Curve": f"{yield_spread:.2f}%",
            "Liquidity Status": lstat
        }
    except:
        return {"Error": "Breadth unavailable"}

def explain_technicals(rsi, macd, sma50, sma200, price):
    ex = {}
    if rsi>70: ex["RSI"] = f"🔴 Overbought ({rsi:.1f}) – may pull back."
    elif rsi<30: ex["RSI"] = f"🟢 Oversold ({rsi:.1f}) – may bounce."
    else: ex["RSI"] = f"🟡 Neutral ({rsi:.1f})."
    if macd['MACD'] > macd['Signal']:
        ex["MACD"] = "🟢 Bullish momentum."
    else:
        ex["MACD"] = "🔴 Bearish momentum."
    if price > sma50:
        ex["SMA50"] = f"🟢 Price above 50-day SMA (${sma50:.2f})."
    else:
        ex["SMA50"] = f"🔴 Price below 50-day SMA (${sma50:.2f})."
    if price > sma200:
        ex["SMA200"] = f"🟢 Price above 200-day SMA (${sma200:.2f})."
    else:
        ex["SMA200"] = f"🔴 Price below 200-day SMA (${sma200:.2f})."
    return ex

def run_analysis(ticker):
    ticker = ticker.upper()
    stock = yf.Ticker(ticker)
    df = stock.history(period="6mo")
    if df.empty:
        return {"Error": "Ticker not found."}
    spy = yf.Ticker("SPY").history(period="6mo")
    if spy.empty:
        spy = yf.download("SPY", period="6mo")  # fallback
        if spy.empty:
            spy = None
    price = df['Close'].iloc[-1]
    df.ta.rsi(append=True)
    df.ta.macd(append=True)
    df.ta.sma(length=50, append=True)
    df.ta.sma(length=200, append=True)
    rsi = df['RSI_14'].iloc[-1]
    macd = {
        'MACD': df['MACD_12_26_9'].iloc[-1],
        'Signal': df['MACDs_12_26_9'].iloc[-1],
        'Hist': df['MACDh_12_26_9'].iloc[-1]
    }
    sma50 = df['SMA_50'].iloc[-1]
    sma200 = df['SMA_200'].iloc[-1]
    support = df['Low'].rolling(60).min().iloc[-1]
    resistance = df['High'].rolling(60).max().iloc[-1]
    info = stock.info
    pe = info.get('trailingPE','N/A')
    eps = info.get('trailingEps','N/A')
    div = info.get('dividendYield','N/A')
    mcap = info.get('marketCap','N/A')
    if mcap!='N/A': mcap = f"${mcap/1e9:.2f}B"

    fig, ax = plt.subplots(figsize=(10,5))
    t_norm = (df['Close']/df['Close'].iloc[0])*100
    if spy is not None and not spy.empty:
        s_norm = (spy['Close']/spy['Close'].iloc[0])*100
        ax.plot(s_norm, label='S&P 500', color='red', ls='--', lw=2)
    ax.plot(t_norm, label=ticker, color='blue', lw=2)
    ax.axhline(100, color='gray', ls=':', alpha=0.5)
    ax.set_title(f'{ticker} vs S&P 500 (Normalized)')
    ax.legend(); ax.grid(True, alpha=0.3)
    plt.tight_layout()
    buf = io.BytesIO()
    plt.savefig(buf, format='png')
    buf.seek(0)
    img = base64.b64encode(buf.getvalue()).decode()
    plt.close(fig)
    plot_url = f"data:image/png;base64,{img}"

    expl = explain_technicals(rsi, macd, sma50, sma200, price)
    macro = get_macro_data()
    breadth = get_breadth_liquidity()
    try:
        news = [n['title'] for n in stock.news[:5]]
    except:
        news = ["News not available."]

    score = 0
    if rsi<30: score+=1
    if rsi>70: score-=1
    if macd['MACD']>macd['Signal']: score+=1
    else: score-=1
    if price>sma50: score+=0.5
    else: score-=0.5
    if price>sma200: score+=0.5
    else: score-=0.5
    if score>=2.5: rec = "🚀 STRONG BUY"
    elif score>=1: rec = "📈 BUY"
    elif score>=-0.5: rec = "✋ HOLD"
    elif score>=-2: rec = "📉 SELL"
    else: rec = "💀 STRONG SELL"

    return {
        "Ticker": ticker,
        "Current Price": round(price,2),
        "RSI": round(rsi,2),
        "MACD": macd,
        "SMA50": round(sma50,2),
        "SMA200": round(sma200,2),
        "Support": round(support,2),
        "Resistance": round(resistance,2),
        "P/E Ratio": pe,
        "EPS": eps,
        "Dividend Yield": div,
        "Market Cap": mcap,
        "Explanations": expl,
        "Plot": plot_url,
        "Macro": macro,
        "Breadth": breadth,
        "News": news,
        "Recommendation": rec
    }
